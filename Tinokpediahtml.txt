<script>
        document.addEventListener('DOMContentLoaded', () => {
            // Date label logic
            const radioLastPeriod = document.getElementById('calcLastPeriod');
            const radioDueDate = document.getElementById('calcDueDate');
            const dateLabel = document.getElementById('dateLabel');

            function updateLabel() {
                if (radioLastPeriod.checked) {
                    dateLabel.textContent = 'תאריך וסת אחרון';
                } else if (radioDueDate.checked) {
                    dateLabel.textContent = 'תאריך לידה משוער';
                }
            }

            if(radioLastPeriod && radioDueDate) {
                radioLastPeriod.addEventListener('change', updateLabel);
                radioDueDate.addEventListener('change', updateLabel);
            }

            // Tab logic
            const tabBtns = document.querySelectorAll('.tab-btn');
            const tabContents = document.querySelectorAll('.tab-content');
            const mainSubmitBtn = document.getElementById('main-submit-btn');
            const googleBtnText = document.getElementById('google-btn-text');

            tabBtns.forEach(btn => {
                btn.addEventListener('click', () => {
                    // Remove active class from all buttons and contents
                    tabBtns.forEach(b => b.classList.remove('active'));
                    tabContents.forEach(c => c.classList.remove('active'));

                    // Add active class to clicked button and target content
                    btn.classList.add('active');
                    const targetId = btn.getAttribute('data-target');
                    document.getElementById(targetId).classList.add('active');

                    // Update button texts
                    if (targetId === 'login-tab') {
                        mainSubmitBtn.innerHTML = 'התחברי <span class="material-symbols-outlined">arrow_back</span>';
                        googleBtnText.textContent = 'התחברות עם Google';
                    } else {
                        mainSubmitBtn.innerHTML = 'בואי נתחיל <span class="material-symbols-outlined">arrow_back</span>';
                        googleBtnText.textContent = 'התחברות מהירה עם Google';
                    }
                });
            });
        });
    </script>
